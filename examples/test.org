#+begin_src R -n :results graphics :file ./test.png :exports results :width 800 :height  600
library(splines)
n <- 1000
x <- runif(n, 0, 1)
doppler <- function(x) {
  sin(20 / (x + 0.2))
}
ytrue <- doppler(x)
epsilon <- rnorm(n, 0, 0.1)
y <- ytrue + epsilon
xseq <- seq(0, 1, length.out = n)
yseq <- doppler(xseq)
plot(x, y, col = rgb(0.4, 0.1, 0.1, 0.5), pch = 19, xlab = "x", ylab = "y")
lines(xseq, yseq, col = "blue", lwd = 2)
nk <- 100
kseq <- seq(0, 1, length.out = nk)
xmat <- spline.des(kseq, x, ord = 4, outer.ok = TRUE)$design
lm1 <- lm(y ~ xmat - 1)
beta <- as.matrix(coef(lm1), ncol = 1)
xseqmat <- spline.des(kseq, xseq, ord = 4, outer.ok = TRUE)$design
yhat <- xseqmat %*% beta
lines(xseq, yhat, col = "red", lwd = 2)
#+end_src
