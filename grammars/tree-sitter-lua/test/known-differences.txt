# A quoted string containing `--[[` (`' --[['`, `'--[['`): the old scanner
# opens a block comment inside the string, and one string node swallows the
# code after it.
*-source/3rd/lovr-api/api/serpent.lua              # --[[ inside a quoted string
*-source/script/parser/compile.lua                 # --[[ inside a quoted string

# A comment that is only dashes: the old grammar nests the next line's
# comment inside a bare `--`, and splits `----` into two comments; here
# each is one comment whose content is the rest of its line.
*-source/meta/3rd/OpenResty/library/resty/websocket/client.lua  # bare or all-dash comment
*-source/lua/nvim-treesitter/utils.lua             # bare or all-dash comment
