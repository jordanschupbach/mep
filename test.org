An exec-gui block runs a program with a window of its own, and shows that
window inside the results: Enter or a click hands it the keyboard, Ctrl-\
takes it back, and when it closes its last picture is kept. This one is
xeyes, written in tkinter -- the pupils follow the mouse pointer:

#+begin_src python :results exec-gui :rows 12
import math
import tkinter as tk

root = tk.Tk()
root.title("xeyes")
canvas = tk.Canvas(root, bg="white", highlightthickness=0)
canvas.pack(fill="both", expand=True)

eyes = []  # (centre x, centre y, half width, half height, pupil item)


def layout(event=None):
    """Draw two eyes side by side, filling the window."""
    global eyes
    canvas.delete("all")
    eyes = []
    w, h = canvas.winfo_width(), canvas.winfo_height()
    rx, ry = w * 0.23, h * 0.45  # an eye's half width and half height
    line = max(3, min(w, h) * 0.04)
    for cx in (w * 0.27, w * 0.73):
        cy = h / 2
        canvas.create_oval(cx - rx, cy - ry, cx + rx, cy + ry,
                           fill="white", outline="black", width=line)
        pupil = canvas.create_oval(0, 0, 0, 0, fill="black", outline="black")
        eyes.append((cx, cy, rx, ry, pupil))
    look()


def look(event=None):
    """Point each pupil at the mouse, keeping it inside its eye."""
    px, py = root.winfo_pointerxy()
    ox, oy = canvas.winfo_rootx(), canvas.winfo_rooty()
    for cx, cy, rx, ry, pupil in eyes:
        pr = min(rx, ry) * 0.3  # pupil radius
        # Pupil centre moves inside an ellipse smaller than the eye by the
        # pupil's radius and the eye's outline.
        ax, ay = max(1.0, rx - pr * 1.6), max(1.0, ry - pr * 1.6)
        dx, dy = px - ox - cx, py - oy - cy
        d = math.hypot(dx / ax, dy / ay)
        if d > 1:
            dx, dy = dx / d, dy / d
        x, y = cx + dx, cy + dy
        canvas.coords(pupil, x - pr, y - pr, x + pr, y + pr)


def watch():
    look()
    root.after(40, watch)


canvas.bind("<Configure>", layout)
watch()
root.mainloop()
#+end_src

#+begin_src python :results exec-gui :rows 10
import math
import tkinter as tk

root = tk.Tk()
root.title("smile")
canvas = tk.Canvas(root, bg="black", highlightthickness=0)
canvas.pack(fill="both", expand=True)

def draw(event=None):
    canvas.delete("all")
    w, h = canvas.winfo_width(), canvas.winfo_height()
    a = min(w * 0.38, h * 1.6)  # half the smile's width
    b = h * 0.4                 # how deep it curves
    cx, cy = w / 2, h * 0.18
    line = max(4, h * 0.03)

    def lip(x):  # the smile's height at x
        return cy + b * math.sqrt(max(0.0, 1 - ((x - cx) / a) ** 2))

    # The tongue, out of the smile halfway to its right end, leaning that way.
    tx = cx + a / 2
    tw, tl = h * 0.24, h * 0.34
    lean = tw * 0.35
    left, right = tx - tw / 2, tx + tw / 2
    pts = [left, lip(left), left, lip(left),
           left + lean * 0.2, lip(tx) + tl * 0.7,
           tx + lean * 0.6, lip(tx) + tl,
           right + lean, lip(tx) + tl * 0.62,
           right, lip(right), right, lip(right)]
    canvas.create_polygon(pts, smooth=True, fill="white", outline="white")
    canvas.create_line(tx + lean * 0.05, lip(tx) + line * 1.5, tx + lean * 0.35, lip(tx) + tl * 0.55,
                       fill="black", width=max(2, line * 0.4), capstyle="round")
    # The smile, drawn over the tongue's root.
    canvas.create_arc(cx - a, cy - b, cx + a, cy + b, start=190, extent=160,
                      style="arc", outline="white", width=line)

canvas.bind("<Configure>", draw)
root.mainloop()
#+end_src

#+begin_definition Orthogonal projection

The point of a subspace \(S\) closest to \(\mathbf{y}\):

\[
\hat{\mathbf{y}} = P\mathbf{y}, \qquad P = X (X^\top X)^{-1} X^\top
\]

- \(P\) is *symmetric*
- \(P^2 = P\)

#+end_definition

#+begin_theorem Pythagoras
If \(\mathbf{a} \perp \mathbf{b}\), the squared lengths
add up: \(\| \mathbf{a} + \mathbf{b} \|^2 = \| \mathbf{a} \|^2 + \| \mathbf{b} \|^2\).

#+end_theorem

#+begin_proof

Expand the square; the cross term is \(2\, \mathbf{a}^\top \mathbf{b} = 0\).

#+begin_remark Nested
Boxes nest, each closing before the one around it.
#+end_remark

#+end_proof
